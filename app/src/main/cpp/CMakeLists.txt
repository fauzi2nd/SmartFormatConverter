
cmake_minimum_required(VERSION 3.22.1)

project(
        smartformatconverter_native
        LANGUAGES C CXX ASM
)

# ==========================================
# ANDROID NATIVE BUILD CONFIGURATION
# ==========================================

if(NOT ANDROID)
    message(FATAL_ERROR
            "This native module is intended for Android only."
    )
endif()

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(CMAKE_POSITION_INDEPENDENT_CODE ON)

# ==========================================
# LIBJPEG-TURBO CONFIGURATION
# ==========================================

# Build the actual libjpeg-turbo library.
# Disable shared variants and optional tools.

set(ENABLE_STATIC ON CACHE BOOL
        "Build static libjpeg-turbo" FORCE
)

set(ENABLE_SHARED OFF CACHE BOOL
        "Disable shared libjpeg-turbo" FORCE
)

set(WITH_TURBOJPEG ON CACHE BOOL
        "Enable TurboJPEG API" FORCE
)

# Preserve the existing CPU-independent build.
set(WITH_SIMD OFF CACHE BOOL
        "Disable SIMD assembly" FORCE
)

# ==========================================
# LIBJPEG-TURBO SOURCE LOCATION
# ==========================================

get_filename_component(
        LIBJPEG_TURBO_DIR
        "${CMAKE_CURRENT_SOURCE_DIR}/../../../../third_party/libjpeg-turbo"
        REALPATH
)

if(NOT EXISTS
        "${LIBJPEG_TURBO_DIR}/CMakeLists.txt"
)
    message(FATAL_ERROR
            "libjpeg-turbo source not found at: "
            "${LIBJPEG_TURBO_DIR}. "
            "Initialize the Git submodule first."
    )
endif()

# ==========================================
# BUILD LIBJPEG-TURBO
# ==========================================

# Do not overwrite CMAKE_SOURCE_DIR.
# Keep libjpeg-turbo isolated in its own
# CMake binary directory.

set(
        LIBJPEG_TURBO_BUILD_DIR
        "${CMAKE_CURRENT_BINARY_DIR}/libjpeg-turbo_build"
)

add_subdirectory(
        "${LIBJPEG_TURBO_DIR}"
        "${LIBJPEG_TURBO_BUILD_DIR}"
        EXCLUDE_FROM_ALL
)

# Verify the expected TurboJPEG static target.

if(NOT TARGET turbojpeg-static)
    message(FATAL_ERROR
            "Required target turbojpeg-static "
            "was not generated by libjpeg-turbo."
    )
endif()

# ==========================================
# JNI JPEG ENCODER
# ==========================================

add_library(
        jpeg_encoder_native
        SHARED
        jpeg_encoder.cpp
)

target_compile_features(
        jpeg_encoder_native
        PRIVATE
        cxx_std_17
)

target_include_directories(
        jpeg_encoder_native
        PRIVATE
        "${LIBJPEG_TURBO_DIR}"
        "${LIBJPEG_TURBO_DIR}/src"
        "${LIBJPEG_TURBO_BUILD_DIR}"
)

# ==========================================
# ANDROID SYSTEM LIBRARIES
# ==========================================

find_library(
        ANDROID_LOG_LIBRARY
        log
        REQUIRED
)

find_library(
        ANDROID_JNIGRAPHICS_LIBRARY
        jnigraphics
        REQUIRED
)

# ==========================================
# LINK NATIVE DEPENDENCIES
# ==========================================

target_link_libraries(
        jpeg_encoder_native
        PRIVATE
        turbojpeg-static
        ${ANDROID_LOG_LIBRARY}
        ${ANDROID_JNIGRAPHICS_LIBRARY}
)

# ==========================================
# ANDROID 16 KB MEMORY PAGE SUPPORT
# ==========================================

# NDK 28+ normally provides compatible
# shared-library ELF alignment.
#
# Explicitly request 16 KB maximum page size
# for the JNI shared library.
#
# This does not modify prebuilt third-party
# libraries such as librxffmpeg-core.so.

if(ANDROID)
    target_link_options(
            jpeg_encoder_native
            PRIVATE
            "-Wl,-z,max-page-size=16384"
    )
endif()

# ==========================================
# BUILD INFORMATION
# ==========================================

message(STATUS
        "SmartFormatConverter native configuration:"
)

message(STATUS
        "  Android ABI: ${ANDROID_ABI}"
)

message(STATUS
        "  Android API: ${ANDROID_PLATFORM}"
)

message(STATUS
        "  libjpeg-turbo: ${LIBJPEG_TURBO_DIR}"
)

message(STATUS
        "  TurboJPEG static target: turbojpeg-static"
)

message(STATUS
        "  SIMD enabled: OFF"
)

message(STATUS
        "  JNI library: jpeg_encoder_native"
)

message(STATUS
        "  Requested ELF page alignment: 16384"
)
